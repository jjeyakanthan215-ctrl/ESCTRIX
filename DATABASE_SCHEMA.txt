================================================================================
                  CIPHER6 (HEXACONNECT) — DATABASE SCHEMA
================================================================================

Database Engine: SQLite (server-side)
Password Hashing: Argon2id (memory=65536, time=3, parallelism=4)

================================================================================
  TABLE 1: users
================================================================================

CREATE TABLE users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    display_name TEXT NOT NULL,
    bio TEXT DEFAULT '',
    avatar_index INTEGER DEFAULT 0,
    password_hash TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_seen TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_username ON users(username);
CREATE INDEX idx_display_name ON users(display_name);

COLUMN DETAILS:
  id             — Auto-incrementing internal ID (never exposed to users)
  username       — Unique identifier, 3-20 chars, lowercase alphanumeric + underscore
  display_name   — Human-readable name, can be changed anytime
  bio            — Short user description, default empty
  avatar_index   — Index into preset avatar list (0-based)
  password_hash  — Argon2id hash of user's password
  created_at     — Account creation timestamp
  last_seen      — Last activity timestamp (updated on login, heartbeat)

================================================================================
  TABLE 2: friends
================================================================================

CREATE TABLE friends (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    friend_id INTEGER NOT NULL,
    status TEXT NOT NULL DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (friend_id) REFERENCES users(id),
    UNIQUE(user_id, friend_id)
);

CREATE INDEX idx_friends_user ON friends(user_id, status);
CREATE INDEX idx_friends_friend ON friends(friend_id, status);

COLUMN DETAILS:
  id             — Auto-incrementing request ID
  user_id        — The user who SENT the request
  friend_id      — The user who RECEIVED the request
  status         — 'pending' | 'accepted' | 'blocked'
  created_at     — When request was sent
  updated_at     — When status was last changed

HOW FRIENDSHIP WORKS:
  - When User A sends request to User B:
      INSERT (user_id=A, friend_id=B, status='pending')

  - When User B accepts:
      UPDATE status='accepted' WHERE user_id=A AND friend_id=B
      INSERT (user_id=B, friend_id=A, status='accepted')
      (Two rows = mutual friendship)

  - When either removes friend:
      DELETE WHERE (user_id=A AND friend_id=B)
      DELETE WHERE (user_id=B AND friend_id=A)
      (Both rows removed)

================================================================================
  TABLE 3: pending_messages
================================================================================

CREATE TABLE pending_messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    sender_id INTEGER NOT NULL,
    recipient_id INTEGER NOT NULL,
    encrypted_content TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sender_id) REFERENCES users(id),
    FOREIGN KEY (recipient_id) REFERENCES users(id)
);

COLUMN DETAILS:
  id                — Auto-incrementing message ID
  sender_id         — User who sent the message
  recipient_id      — User who should receive the message
  encrypted_content — AES-256-GCM encrypted message (base64 encoded)
  created_at        — When message was stored

NOTE: The server NEVER sees plaintext messages. The encrypted_content is
encrypted by the sender before being sent to the server. The server is
just a blind mailbox that stores and forwards.

FLOW:
  1. User A sends message to offline User B
  2. Server stores encrypted message in this table
  3. User B logs in
  4. Server delivers all pending messages to User B
  5. Server deletes delivered messages from this table

================================================================================
  TABLE 4: login_attempts (for throttling)
================================================================================

CREATE TABLE login_attempts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT NOT NULL,
    ip_address TEXT NOT NULL,
    attempted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    success BOOLEAN DEFAULT FALSE
);

THROTTLING RULES:
  - Track failed login attempts per username
  - 5 failed attempts within 5 minutes = account locked for 5 minutes
  - Successful login resets the counter

================================================================================
  DATABASE FUNCTIONS
================================================================================

create_user(username, display_name, password)
  → Validate username format and uniqueness
  → Hash password with Argon2id
  → Insert into users table
  → Return user data + JWT token

authenticate(username, password)
  → Find user by username (case-insensitive)
  → Check login throttle
  → Verify Argon2id hash
  → Update last_seen
  → Return user data or error

search_users(query, current_user_id, limit=20)
  → WHERE username LIKE '%query%' OR display_name LIKE '%query%'
  → Exclude current user from results
  → Include friend_status for each result
  → Return list of matching users

send_friend_request(from_user_id, to_user_id)
  → Check not self-request
  → Check no existing relationship
  → Insert with status='pending'
  → Return request ID

accept_friend_request(request_id, accepting_user_id)
  → Verify request exists and is addressed to accepting_user
  → Update status to 'accepted'
  → Create reverse entry (mutual friendship)
  → Return success

decline_friend_request(request_id, declining_user_id)
  → Verify request exists and is addressed to declining_user
  → Delete the pending request
  → Return success

remove_friend(user_id, friend_id)
  → Delete both direction entries
  → Return success

get_friends(user_id)
  → SELECT all users where friendship status = 'accepted'
  → Include online/offline status from connection map
  → Return list sorted: online first, then alphabetical

get_pending_requests(user_id)
  → SELECT all incoming requests where status = 'pending'
  → Return with sender's profile info

get_sent_requests(user_id)
  → SELECT all outgoing requests where status = 'pending'
  → Return with recipient's profile info

are_friends(user_id_1, user_id_2)
  → Check if mutual friendship exists (status = 'accepted')
  → Return boolean (used before allowing calls/messages)

update_profile(user_id, display_name, bio, avatar_index)
  → Update user's profile fields
  → Return updated profile

get_user_profile(username)
  → Find user by username
  → Return full profile data

store_pending_message(sender_id, recipient_id, encrypted_content)
  → Must be mutual friends (server checks)
  → Insert into pending_messages table

get_pending_messages(user_id)
  → SELECT all messages for this recipient
  → DELETE retrieved messages
  → Return list of messages with sender info
